  <script>
  // Smooth scroll
  document.querySelectorAll('nav ul li a').forEach(anchor => {
    anchor.addEventListener('click', function(e) {
      e.preventDefault();
      document.querySelector(this.getAttribute('href')).scrollIntoView({ behavior: 'smooth' });
    });
  });

  // Login / Register Tab Switch
  const tabSignIn = document.getElementById('tabSignIn');
  const tabRegister = document.getElementById('tabRegister');
  const signInForm = document.getElementById('signInForm');
  const registerForm = document.getElementById('registerForm');
  const heroCreateBtn = document.getElementById('heroCreateBtn');

  if (tabSignIn && tabRegister) {
    tabRegister.addEventListener('click', () => {
      signInForm.classList.add('hidden');
      registerForm.classList.remove('hidden');
      tabRegister.classList.remove('text-gray-400');
      tabRegister.classList.add('border-b-2','border-black','font-semibold');
      tabSignIn.classList.add('text-gray-400');
      tabSignIn.classList.remove('border-b-2','border-black','font-semibold');
    });

    tabSignIn.addEventListener('click', () => {
      registerForm.classList.add('hidden');
      signInForm.classList.remove('hidden');
      tabSignIn.classList.remove('text-gray-400');
      tabSignIn.classList.add('border-b-2','border-black','font-semibold');
      tabRegister.classList.add('text-gray-400');
      tabRegister.classList.remove('border-b-2','border-black','font-semibold');
    });
  }

  // Hero Create Account -> Open Register Tab
  if(heroCreateBtn){
    heroCreateBtn.addEventListener('click', () => {
      setTimeout(() => {
        tabRegister.click();
      }, 300);
    });
  }
</script>
